cmake_minimum_required(VERSION 3.21)
project(TelegramEmoji LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Test)
file(GLOB CODEGEN codegen/common/*.cpp codegen/emoji/*.cpp)
add_executable(codegen_emoji ${CODEGEN})
target_include_directories(codegen_emoji PRIVATE .)
target_link_libraries(codegen_emoji PRIVATE Qt6::Core Qt6::Gui)
target_precompile_headers(codegen_emoji PRIVATE <algorithm> <cstring> <memory>)
add_library(telegram_emoji engine/emoji_suggestions.cpp generated/emoji_suggestions_data.cpp)
target_include_directories(telegram_emoji PUBLIC engine generated)
target_precompile_headers(telegram_emoji PRIVATE <cstring>)
add_executable(emoji_probe probe.cpp)
target_link_libraries(emoji_probe PRIVATE telegram_emoji Qt6::Core)
