cmake_minimum_required(VERSION 3.21)
project(LanLinkCpp VERSION 0.6.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network Sql Test Concurrent)
option(LANLINK_FULL_TELEGRAM_UI "Use original complete Telegram lib_ui InputField" OFF)
if(LANLINK_FULL_TELEGRAM_UI)
 if(NOT TARGET desktop-app::lib_ui)
  message(FATAL_ERROR "Configure via ../full-ui-probe with LANLINK_BUILD_APPLICATION=ON for full UI")
 endif()
 set(CMAKE_CXX_STANDARD 20)
 add_library(telegram_emoji INTERFACE)
 target_include_directories(telegram_emoji INTERFACE telegram/emoji)
 target_link_libraries(telegram_emoji INTERFACE desktop-app::lib_ui)
 set(input_source telegram/telegram_full_input_adapter.cpp telegram/telegram_full_input_adapter.h)
else()
 add_library(telegram_emoji telegram/emoji/emoji_suggestions.cpp telegram/emoji/emoji_suggestions_data.cpp)
 target_include_directories(telegram_emoji PUBLIC telegram/emoji)
 target_precompile_headers(telegram_emoji PRIVATE <cstring>)
 set(input_source telegram/telegram_input_adapter.cpp)
endif()
set(LANLINK_ICON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../packaging/icons")
add_library(lanlink_miniz STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tdef.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_zip.c)
target_include_directories(lanlink_miniz PUBLIC third_party/miniz)
target_compile_definitions(lanlink_miniz PUBLIC MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
set_target_properties(lanlink_miniz PROPERTIES C_STANDARD 99 AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
add_library(lanlink_core src/contactmailbook.cpp src/chatmailexport.cpp telegram/telegram_history_presentation.cpp telegram/telegram_row_painter.cpp telegram/telegram_input_logic.cpp ${input_source} src/node.cpp src/networkpolicy.cpp src/filetransfer.cpp src/transfercard.cpp src/screenservice.cpp src/linkpreview.cpp src/chatnotifications.cpp src/emojipicker.cpp src/updateversion.cpp src/updaterelease.cpp src/updatezip.cpp src/updatestaging.cpp src/updatecredentials.cpp src/updateapply.cpp src/updateservice.cpp src/updateicon.cpp)
if(LANLINK_FULL_TELEGRAM_UI)
 target_compile_definitions(lanlink_core PUBLIC LANLINK_FULL_TELEGRAM_UI)
 target_sources(lanlink_core PRIVATE telegram/telegram_message_text.cpp telegram/telegram_message_text.h telegram/telegram_dialogs_inner.cpp telegram/telegram_lan_history_list.cpp telegram/telegram_transfer_list.cpp telegram/telegram_sharing_box.cpp)
 target_compile_features(lanlink_core PUBLIC cxx_std_20)
 target_precompile_headers(lanlink_core PRIVATE <ui/ui_pch.h>)
 # Compile original Telegram dialog control styles and original icon masks.
 generate_styles(lanlink_core "${CMAKE_CURRENT_SOURCE_DIR}/telegram"
     "lan_original_dialogs.style" "${upstream}/Telegram/lib_ui/ui/widgets/widgets.style")
 target_link_directories(lanlink_core PUBLIC /tmp/telegram-sysroot/usr/lib/x86_64-linux-gnu)
endif()
target_include_directories(lanlink_core PUBLIC src telegram)
target_link_libraries(lanlink_core PUBLIC telegram_emoji lanlink_miniz Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql Qt6::Widgets Qt6::Concurrent)
target_compile_definitions(lanlink_core PUBLIC LANLINK_VERSION="${PROJECT_VERSION}")
if(WIN32)
 target_link_libraries(lanlink_core PRIVATE advapi32)
endif()
# Mail is intentionally independent of LAN discovery, identity and transfer state.
add_library(lanlink_mail STATIC src/mailaccounts.cpp src/mailoauth.cpp src/mailoauth.h src/mailtransport.cpp src/mailmime.cpp src/mailsyncstate.cpp src/mailwindow.cpp src/mailwindow.h)
target_include_directories(lanlink_mail PUBLIC src)
target_link_libraries(lanlink_mail PUBLIC Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Network)
if(LANLINK_FULL_TELEGRAM_UI)
 target_sources(lanlink_mail PRIVATE telegram/telegram_mail_widgets.cpp)
 target_link_libraries(lanlink_mail PUBLIC lanlink_core)
 target_precompile_headers(lanlink_mail PRIVATE <ui/ui_pch.h>)
endif()
if(WIN32)
 target_link_libraries(lanlink_mail PRIVATE advapi32)
endif()
# Compile the production window once for the app and integration tests.
add_library(lanlink_window STATIC src/mainwindow.cpp src/mainwindow.h)
target_link_libraries(lanlink_window PUBLIC lanlink_core lanlink_mail)
add_executable(LanLink WIN32 src/main.cpp)
target_link_libraries(LanLink PRIVATE lanlink_window)
set_target_properties(LanLink PROPERTIES AUTORCC ON)
target_sources(LanLink PRIVATE "${LANLINK_ICON_DIR}/lanlink.qrc")
if(WIN32)
 target_sources(LanLink PRIVATE "${LANLINK_ICON_DIR}/lanlink.rc")
 set_source_files_properties("${LANLINK_ICON_DIR}/lanlink.rc" PROPERTIES INCLUDE_DIRECTORIES "${LANLINK_ICON_DIR}")
endif()
if(MSVC)
 target_compile_options(lanlink_core PRIVATE /utf-8)
 target_compile_options(lanlink_mail PRIVATE /utf-8)
 target_compile_options(lanlink_window PRIVATE /utf-8)
 target_compile_options(LanLink PRIVATE /utf-8)
endif()
enable_testing()
file(GLOB TEST_SOURCES CONFIGURE_DEPENDS tests/*.cpp)
foreach(source ${TEST_SOURCES})
 get_filename_component(testname ${source} NAME_WE)
 add_executable(${testname} ${source})
 target_link_libraries(${testname} PRIVATE lanlink_core Qt6::Test)
 if(testname MATCHES "^test_mail")
  target_link_libraries(${testname} PRIVATE lanlink_mail)
 endif()
 if(MSVC)
  target_compile_options(${testname} PRIVATE /utf-8)
 endif()
 add_test(NAME ${testname} COMMAND ${testname})
 set_tests_properties(${testname} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 90)
endforeach()
if(TARGET test_ui)
 target_link_libraries(test_ui PRIVATE lanlink_window)
endif()
if(TARGET test_networkdiagnosticsui)
 target_link_libraries(test_networkdiagnosticsui PRIVATE lanlink_window)
endif()
if(TARGET test_transfer_timeline)
 target_link_libraries(test_transfer_timeline PRIVATE lanlink_window)
endif()
if(TARGET test_notification_transport)
 target_link_libraries(test_notification_transport PRIVATE lanlink_window)
endif()
add_executable(interop_peer tools/interop_peer.cpp)
target_link_libraries(interop_peer PRIVATE lanlink_core)
if(MSVC)
 target_compile_options(interop_peer PRIVATE /utf-8)
endif()

if(LANLINK_FULL_TELEGRAM_UI)
 # Resource objects must be linked into executables, not discarded from static archives.
 get_property(all_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
 foreach(t ${all_targets})
  get_target_property(kind ${t} TYPE)
  if(kind STREQUAL "EXECUTABLE")
   set_target_properties(${t} PROPERTIES AUTORCC ON)
   target_sources(${t} PRIVATE ${emoji_qrc})
  endif()
 endforeach()
endif()
