cmake_minimum_required(VERSION 3.21)
project(TelegramInputAdapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Test)
add_library(telegram_input telegram_input_logic.cpp telegram_input_adapter.cpp telegram_input_adapter.h)
target_include_directories(telegram_input PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(telegram_input PUBLIC Qt6::Widgets)
add_executable(test_input test_input.cpp)
target_link_libraries(test_input PRIVATE telegram_input Qt6::Test)
enable_testing()
add_test(NAME input COMMAND test_input)
set_tests_properties(input PROPERTIES ENVIRONMENT QT_QPA_PLATFORM=offscreen)
