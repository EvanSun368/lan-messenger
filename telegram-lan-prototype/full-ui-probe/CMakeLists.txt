cmake_minimum_required(VERSION 3.25)
project(TelegramFullUiProbe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
# This integration host has only been implemented and validated for Linux.
# Do not force Telegram's Linux branches on a Windows/macOS toolchain.
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
 message(FATAL_ERROR "The full Telegram integration host is Linux-only at this stage. A native dependency/build port is required for Windows or macOS; do not use a default-mode binary as proof of full-mode support.")
endif()
set(LINUX TRUE)
set(DESKTOP_APP_USE_PACKAGED ON CACHE BOOL "")
set(DESKTOP_APP_USE_PACKAGED_FONTS ON CACHE BOOL "")
set(DESKTOP_APP_DISABLE_CRASH_REPORTS ON CACHE BOOL "")
set(DESKTOP_APP_DISABLE_AUTOUPDATE ON CACHE BOOL "")
set(DESKTOP_APP_DISABLE_QT_PLUGINS ON CACHE BOOL "")
set(DESKTOP_APP_USE_PANGO OFF CACHE BOOL "")
set(CRL_FORCE_QT ON CACHE BOOL "")
get_filename_component(upstream ../upstream REALPATH)
set(third_party_loc ${upstream}/Telegram/ThirdParty)
set(submodules_loc ${upstream}/Telegram)
set(cmake_helpers_loc ${upstream}/cmake)
include(cmake/variables.cmake)
include(cmake/nice_target_sources.cmake)
include(cmake/target_compile_options_if_exists.cmake)
include(cmake/target_link_frameworks.cmake)
include(cmake/target_link_options_if_exists.cmake)
include(cmake/init_target.cmake)
include(cmake/generate_target.cmake)
include(cmake/target_prepare_qrc.cmake)
include(cmake/options.cmake)
include(cmake/external/qt/package.cmake)
set(desktop_app_skip_libs ada angle auto_updates boost cld3 cmark_gfm dispatch ffmpeg glibmm hunspell iconv microtex minizip openal openh264 opus pango qr_code_generator rnnoise tde2e tdesktop_rust tlottie tmc ton variant vpx wallet_engine webrtc)
add_subdirectory(cmake)
add_subdirectory(${upstream}/Telegram/lib_rpl lib_rpl)
add_subdirectory(${upstream}/Telegram/lib_crl lib_crl)
add_subdirectory(${upstream}/Telegram/lib_base lib_base)
add_subdirectory(${upstream}/Telegram/codegen codegen)
add_subdirectory(${upstream}/Telegram/lib_ui lib_ui)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(input_field_probe input_field_probe.cpp)
target_link_libraries(input_field_probe PRIVATE desktop-app::lib_ui Qt6::Test)
target_include_directories(lib_ui SYSTEM PRIVATE /tmp/telegram-sysroot/usr/include/x86_64-linux-gnu)
target_link_directories(input_field_probe PRIVATE /tmp/telegram-sysroot/usr/lib/x86_64-linux-gnu)
set_target_properties(input_field_probe PROPERTIES AUTORCC ON)
file(GLOB emoji_qrc "${upstream}/Telegram/Resources/qrc/emoji_*.qrc")
target_sources(input_field_probe PRIVATE ${emoji_qrc})

option(LANLINK_BUILD_APPLICATION "Build LanLink with original InputField" OFF)
if(LANLINK_BUILD_APPLICATION)
 set(LANLINK_FULL_TELEGRAM_UI ON CACHE BOOL "" FORCE)
 add_subdirectory(../application application)
endif()
