cmake_minimum_required(VERSION 3.21)
project(TelegramDialogsAdapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Test)
add_library(telegram_dialogs_adapter conversation_list.cpp conversation_list.h telegram_dialogs_model.h telegram_row_painter.cpp telegram_row_painter.h)
target_include_directories(telegram_dialogs_adapter PUBLIC .)
target_link_libraries(telegram_dialogs_adapter PUBLIC Qt6::Widgets)
add_executable(test_telegram_dialogs test_dialogs.cpp)
target_link_libraries(test_telegram_dialogs PRIVATE telegram_dialogs_adapter Qt6::Test)
enable_testing()
add_test(NAME telegram_dialogs COMMAND test_telegram_dialogs)
set_tests_properties(telegram_dialogs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
