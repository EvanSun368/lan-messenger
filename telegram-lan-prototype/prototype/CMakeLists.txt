cmake_minimum_required(VERSION 3.21)
project(TelegramLanSpike LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network Sql Test)
add_library(lan_node backend/node.cpp backend/node.h)
target_include_directories(lan_node PUBLIC backend)
target_link_libraries(lan_node PUBLIC Qt6::Core Qt6::Gui Qt6::Network Qt6::Sql)
add_executable(TelegramLanSpike src/main.cpp)
target_include_directories(TelegramLanSpike PRIVATE vendor src)
target_link_libraries(TelegramLanSpike PRIVATE lan_node Qt6::Widgets)
add_executable(test_spike src/test_spike.cpp)
target_include_directories(test_spike PRIVATE vendor src)
target_link_libraries(test_spike PRIVATE lan_node Qt6::Widgets Qt6::Test)
enable_testing()
add_test(NAME spike COMMAND test_spike)
set_tests_properties(spike PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
